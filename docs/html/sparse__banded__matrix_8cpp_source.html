5matrix<T> matrix<T>::banded(int N, int kl, int ku, T val)
6{
7 // Estimated memory (over-estimation) for reservation
8 MKL_INT estimated_nnz = N * (ku + kl + 1);
9
10 std::vector<T> values;
11 std::vector<MKL_INT> col_idx;
12 std::vector<MKL_INT> row_idx(N+1);
13
14 // Reserve memory
15 values.reserve(estimated_nnz);
16 col_idx.reserve(estimated_nnz);
17
18 int current_nnz_count = 0;
19
20 for (int i = 0; i < N; i++)
21 {
22 row_idx[i] = current_nnz_count;
23
24 int start_col = std::max(0, i - kl);
25 int end_col = std::min(N - 1, i + ku);
26
27 for (int j = start_col; j <= end_col; j++)
28 {
29 values.push_back(val);
30 col_idx.push_back(j);
31 current_nnz_count++;
32 }
33 }
34
35 row_idx[N] = current_nnz_count; // The last element of the row_idx vector in CSR must always be nnz,
36 // i.e. the size of the value vector
37
38 // Shrink the over-estimated reserved memory
39 values.shrink_to_fit();
40 col_idx.shrink_to_fit();
41
42 return matrix<T>(values, col_idx, row_idx, N, N);
43}
